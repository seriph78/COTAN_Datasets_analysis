getClData <- function(objCOTAN, clList, cl) {
  checker <- new("SimpleGDIUniformityCheck")
  cluster <- names(clList)[[cl]]
  cells <- clList[[cl]]
  res <- checkClusterUniformity(objCOTAN, cluster = cluster, cells = cells,checker = checker, 
                                 cores = 6, saveObj = FALSE)
  return(res)
}
