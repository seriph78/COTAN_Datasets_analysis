Extract loom data to SCE objects

readLoomFile <- function(loomFile, chunkSize = 10000L) {
  h5 <- H5File$new(loomFile, mode = "r")
  on.exit(h5$close_all(), add = TRUE)

  ## --- row / col metadata (small; read fully) ---

  rowAttrsGrp  <- h5[["row_attrs"]]
  rowAttrNames <- names(rowAttrsGrp)
  rowMetaDf <- as.data.frame(
    lapply(rowAttrNames, function(nm) rowAttrsGrp[[nm]][]),
    col.names        = rowAttrNames,
    stringsAsFactors = FALSE
  )

  colAttrsGrp  <- h5[["col_attrs"]]
  colAttrNames <- names(colAttrsGrp)
  colMetaDf <- as.data.frame(
    lapply(colAttrNames, function(nm) colAttrsGrp[[nm]][]),
    col.names        = colAttrNames,
    stringsAsFactors = FALSE
  )

  nGenesMeta <- nrow(rowMetaDf)
  nCellsMeta <- nrow(colMetaDf)
  
 ## --- global attrs ---

  attrsGrp  <- h5[["attrs"]]
  attrNames <- names(attrsGrp)
  globalAttrs <- lapply(attrNames, function(nm) attrsGrp[[nm]][])
  names(globalAttrs) <- attrNames

  ## --- resd matrix via true chunking on rows: dense -> sparse -> rbind ---

  if (str_ends(loomFile, "SS2.loom")) {
    ds   <- h5[["layers"]][["estimated_counts"]]
  } else {
    ds   <- h5[["matrix"]]
  }
  dims <- ds$get_space()$get_simple_extent_dims()$dims
  nRow <- dims[1]
  nCol <- dims[2]

  allCols <- seq_len(nCol)

  iList <- list()
  jList <- list()
  xList <- list()

  for (startRow in seq(1L, nRow, by = chunkSize)) {
    endRow <- min(startRow + chunkSize - 1L, nRow)

    print(paste0("Handling rows: ", str_pad(paste0(startRow), 5),
                 " - ", str_pad(paste0(endRow), 5)))

    chunk <- ds[startRow:endRow, allCols]      # only subset of rows, full cols
    nz    <- which(chunk != 0, arr.ind = TRUE)
    if (nrow(nz) == 0L) {
      rm(chunk, nz); next
    }

    globalRow <- nz[, 1] + (startRow - 1L)     # map to [1..nRow]
    globalCol <- nz[, 2]                       # already [1..nCol]
    vals      <- chunk[nz]

    iList[[startRow]] <- globalRow
    jList[[startRow]] <- globalCol
    xList[[startRow]] <- as.numeric(vals)

    rm(chunk, nz, globalRow, globalCol, vals)
    gc()
  }

  i <- do.call(c, iList)
  j <- do.call(c, jList)
  x <- do.call(c, xList)

  print("Creating the sparse matrix")

  countsFileOrder <- sparseMatrix(
    i    = i,
    j    = j,
    x    = x,
    dims = c(nRow, nCol)
  )

  rm(iList, jList, xList, i, j, x)
  gc()

  ## --- orient to genes x cells ---
  if (nRow == nGenesMeta && nCol == nCellsMeta) {
    countsGenesByCells <- countsFileOrder
  } else if (nRow == nCellsMeta && nCol == nGenesMeta) {
    countsGenesByCells <- Matrix::t(countsFileOrder)
  } else {
    stop("Cannot match matrix dims to metadata in: ", loomFile)
  }

  rownames(countsGenesByCells) <- make.unique(rowMetaDf[["gene_names"]])
  colnames(countsGenesByCells) <- make.unique(colMetaDf[["cell_names"]])

  sce <- SingleCellExperiment(
    assays  = list(counts = countsGenesByCells),
    rowData = rowMetaDf,
    colData = colMetaDf
  )

  list(
    sce         = sce,
    globalAttrs = globalAttrs
  )
}

Convert SCE objects to COTAN

## helper: convert a single SCE + its global attrs into a COTAN object
sceToCotan <- function(sce, globalAttrs = list()) {
  # 1. SCE -> COTAN (handles counts, rowData, colData, cluster columns, etc.)
  objCotan <- convertFromSingleCellExperiment(objSCE = sce)

  # 2. Initialize metaDataset with standard tags (use whatever you have)
  #    If your globalAttrs already contain these, pick from there.
  geoVal   <-
    if ("GEO"   %in% names(globalAttrs)) globalAttrs[["GEO"]]   else ""
  seqVal   <- 
    if ("method" %in% names(globalAttrs)) globalAttrs[["method"]] else ""
  condVal  <- 
    if ("condition" %in% names(globalAttrs)) globalAttrs[["condition"]] else ""

  objCotan <- initializeMetaDataset(
    objCotan,
    GEO              = as.character(geoVal),
    sequencingMethod = as.character(seqVal),
    sampleCondition  = as.character(condVal)
  )

  # 3. Add all remaining global attributes as extra metadata rows
  usedTags <- c("GEO", "method", "condition")
  extraTags <- base::setdiff(names(globalAttrs), usedTags)

  for (tag in extraTags) {
    val <- globalAttrs[[tag]]
    # COTAN expects a character vector; coerce anything else
    objCotan <- addElementToMetaDataset(
      objCotan,
      tag   = tag,
      value = as.character(val)
    )
  }

  objCotan
}

Extract and split data to multiple COTAN objects

loadLoomAndSplitByInputAndSaveCotan <- function(loomFile,
                                                outDir,
                                                chunkSize = 10000L) {
  print(paste("Loading Loom file:", loomFile))
  c(sce, globalAttrs) %<-% readLoomFile(loomFile, chunkSize = chunkSize)
  
  globalAttrs <- append(globalAttrs,
                        list(loomName = stringr::str_remove(basename(loomFile),
                                                            pattern = ".loom")))

  print("Creating COTAN object")
  objCOTAN <- sceToCotan(sce, globalAttrs)
  
  rm(sce, globalAttrs)
  gc()

  loomName <- str_remove(getMetadataElement(objCOTAN, "loomName"),
                         pattern = "systematic-comparison-")
  
  if (str_ends(loomName, "SS2")) {
    fileOut <- file.path(outDir, paste0(loomName, ".RDS"))
    
    print(paste("Saving COTAN object as:", fileOut))
    print(paste0(getDims(objCOTAN)))

    saveRDS(objCOTAN, file = fileOut)

#    crossCheck <-
#      table(getMetadataCells(objCOTAN)[["input_name"]])

    inputNames <- unique(getMetadataCells(objCOTAN)[["input_name"]])

#    assert_that(sum(crossCheck != 0) == length(inputNames))

    # Split and save by input name
    for(inputName in inputNames) {
      print(paste("Split by input name:", inputName))
      
      cellsToDrop <-
        getCells(objCOTAN)[getMetadataCells(objCOTAN)[["input_name"]] != inputName]
      
      splitObj <- dropGenesCells(objCOTAN, cells = cellsToDrop)
      
      fileOut <- file.path(outDir, paste0(loomName, "_", inputName, ".RDS"))
      
      print(paste("Saving COTAN object as:", fileOut))
      print(paste0(getDims(splitObj)))
  
      saveRDS(splitObj, file = fileOut)
  
      rm(splitObj)
    }

  } else {
    crossCheck <- 
      table(stringr::str_split_fixed(getCells(objCOTAN),
                                     pattern = "-", n = 2)[, 2],
            getMetadataCells(objCOTAN)[["input_id"]])
    
    inputIDs        <-
      str_split_1(getMetadataElement(objCOTAN, "input_id"),   pattern = ", ")
    names(inputIDs) <-
      str_split_1(getMetadataElement(objCOTAN, "input_name"), pattern = ", ")
    
    assert_that(sum(crossCheck != 0) == length(inputIDs))
    
    
    # Split and save by input name
    for(inputName in names(inputIDs)) {
      print(paste("Split by input name:", inputName))
      
      inputID <- inputIDs[[inputName]]
      cellsToDrop <-
        getCells(objCOTAN)[getMetadataCells(objCOTAN)[["input_id"]] != inputID]
      
      splitObj <- dropGenesCells(objCOTAN, cells = cellsToDrop)
      
      fileOut <- file.path(outDir, paste0(loomName, "_", inputName, ".RDS"))
      
      print(paste("Saving COTAN object as:", fileOut))
      print(paste0(getDims(splitObj)))
  
      saveRDS(splitObj, file = fileOut)
  
      rm(splitObj)
    }
  }

  gc()
  return(NULL)
}
